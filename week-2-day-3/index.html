Simulate API call with delay + Extension: Nested timeout for "Processing Data..." to demonstrate asynchronous flow.

function fakeAPICall() {
    console.log("Fetching user details...");
    setTimeout(() => {
        console.log("User data received ");
        // Extension: Nested timeout
        setTimeout(() => {
            console.log("Processing Data...");
        }, 1000);
    }, 2000);
}

Task 2 – Promise Example (.then & .catch)

Promise banate hain jo isOnline check karta hai. Success hone par resolve aur failure par reject karta hai.

const simulateFetch = new Promise((resolve, reject) => {
    let isOnline = true;
    setTimeout(() => {
        if (isOnline) resolve("Data fetched successfully ");
        else reject("Network error ");
    }, 1500);
});

simulateFetch
    .then(msg => console.log(msg))
    .catch(err => console.error(err));

Task 3 – Async/Await with Fetch

Real JSONPlaceholder API se posts mangwane ke liye async/await aur try...catch ka upyog.

async function loadPosts() {
    try {
        const response = await fetch("https://jsonplaceholder.typicode.com/posts?_limit=5");
        const posts = await response.json();
        console.log("Latest Posts:", posts);
    } catch (error) {
        console.error("Error loading posts:", error);
    }
}

loadPosts();
